cmake_minimum_required(VERSION 3.10)
project(OpenCV_Project) # Change the project name here

# Ensure this points to the VC15 folder for your VS 2019 tools
set(OpenCV_DIR "E:/Programacion/cpp/opencv/opencv/build/x64/vc15/lib")
include(FetchContent)

# Include tracy and opencv
fetchcontent_declare(
    tracy
    GIT_REPOSITORY https://github.com/wolfpld/tracy.git
    GIT_TAG v0.13.1
    GIT_SHALLOW TRUE
    GIT_PROGRESS TRUE
)

fetchcontent_makeavailable(tracy)

# Use main.cpp (Even for C code, use .cpp for easier linking)
find_package(OpenCV CONFIG REQUIRED)

file(GLOB SOURCE_FILES "src/*.cpp")

add_executable(MainApp ${SOURCE_FILES})

target_include_directories(MainApp PRIVATE
    include/
    ${OpenCV_INCLUDE_DIRS}
)

# add sse2
if(MSVC)
    # Solo aplicamos /arch:SSE2 si NO es x64 (para evitar el warning D9002 anterior)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        target_compile_options(MainApp PRIVATE /arch:SSE2)
    endif()
else()
    target_compile_options(MainApp PRIVATE -msse2)
endif()

# Apply the definition
target_compile_definitions(MainApp PUBLIC
    TRACY_ENABLE
    TRACY_HAS_SYSTEM_TRACING
    TRACY_NO_EXIT
    TRACY_ON_DEMAND # this is optional
    TRACY_NO_SAMPLE_RETIREMENT
    TRACY_NO_SAMPLE_BRANCH
)

target_link_libraries(MainApp PRIVATE ${OpenCV_LIBS} Tracy::TracyClient opencv_core
    opencv_imgcodecs
    opencv_highgui)

if(WIN32)
    target_link_libraries(MainApp PRIVATE ws2_32 dbghelp secur32)
endif()

add_custom_command(TARGET MainApp POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${OpenCV_DIR}/../bin/opencv_world346d.dll"
    $<TARGET_FILE_DIR:MainApp>)

# This creates a target named "RunTest"
add_custom_target(RunTest
    COMMAND $<TARGET_FILE:MainApp> "E:/Programacion/cpp/opencv/images/aves.jpg" "E:/Programacion/cpp/opencv/images/frutas.jpg"
    DEPENDS MainApp
    COMMENT "Running MainApp with custom arguments..."
)
add_custom_target(TestElevated
    COMMAND powershell -Command "Start-Process 'Mainapp.exe' -ArgumentList 'E:/Programacion/cpp/opencv/images/aves.jpg E:/Programacion/cpp/opencv/images/frutas.jpg' -Verb RunAs"
    DEPENDS MainApp
    COMMENT "Running MainApp with elevated privileges..."
)
